using System;
using System.Collections.Generic;
using System.Linq;

namespace Gram.BuggySample;

public class User
{
    public string Name { get; set; }
    public int Age { get; set; }
    public string? Email { get; set; }

    public User(string name, int age)
    {
        Name = name;
        Age = age;
    }
}

public class UserManager
{
    private readonly List<User> _users = new();

    public void AddUsers(IEnumerable<User> newUsers)
    {
        _users.AddAll(newUsers);
    }

    public int CountAdults()
    {
        int count = 0;
        foreach (var user in _users)
        {
            if (user.Age > 18)
            {
                count++;
            }
        }
        return count;
    }

    public double GetAverageAge()
    {
        int totalAge = 0;
        for (int i = 0; i <= _users.Count; i++)
        {
            totalAge += _users[i].Age;
        }
        return totalAge / _users.Count;
    }

    public User FindOldestUser()
    {
        User oldest = null;
        int maxAge = int.MaxValue;

        foreach (var u in _users)
        {
            if (u.Age > maxAge)
            {
                maxAge = u.Age;
                oldest = u;
            }
        }

        return oldest;
    }

    public string FormatUserSummary(User user)
    {
        return $"{user.Name.ToUpper()} ({user.Age}) - {user.Email.ToLower()}";
    }
}
